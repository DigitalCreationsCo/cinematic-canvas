/**
 * parseSceneFromText
 * Orchestrates the extraction of characters, locations, and props from raw text,
 * hydrates the knowledge base, and informs scene generation.
 */
async parseSceneFromText(job: Job, sceneFields: Partial<SceneFields>): Promise<Scene> {
  const { projectId, userId, traceId } = job;
  const rawInput = sceneFields.description || "";

  // 1. Extract and Resolve Mentions
  const extractionResult = await this.kbService.extractAndResolveMentions({
    htmlInput: rawInput,
    projectId,
    userId,
  });

  const { handlesResolved, textPlain } = extractionResult;

  // 2. Resolve Existing Entities
  const [existingChars, existingLocs, existingProps, existingScenes] = await Promise.all([
    this.projectRepository.getCharactersByProject(projectId),
    this.projectRepository.getLocationsByProject(projectId),
    this.projectRepository.getPropsByProject(projectId),
    this.projectRepository.getScenesByProject(projectId),
  ]);

  const resolvedCharacters = existingChars.filter((c) => handlesResolved.includes(c.referenceId));
  const resolvedLocations = existingLocs.filter((l) => handlesResolved.includes(l.referenceId));
  const resolvedProps = existingProps.filter((p) => handlesResolved.includes(p.referenceId));

  const toolContext = this.createToolContext(job, traceId);

  // 3. Parse Unresolved Entities from Plain Text
  const [charAttrs, locAttrs, propAttrs] = await Promise.all([
    needsEntityTextParsing(textPlain)
      ? createParseCharactersTool({ context: toolContext }).run({ input: textPlain })
      : Promise.resolve([]),
    needsEntityTextParsing(textPlain)
      ? createParseLocationsTool({ context: toolContext }).run({ input: textPlain })
      : Promise.resolve([]),
    needsEntityTextParsing(textPlain)
      ? createParsePropsTool({ context: toolContext }).run({ input: textPlain })
      : Promise.resolve([]),
  ]);

  // 4. Persistence and Normalization
  const toInsertChars = (charAttrs || []).map(attr => mapDomainCharacterToInsertCharacter({ ...attr, projectId }));
  const toInsertLocs = (locAttrs || []).map(attr => mapDomainLocationToInsertLocation({ ...attr, projectId }));
  const toInsertProps = (propAttrs || []).map(attr => mapDomainPropToInsertProp({ ...attr, projectId }));

  await Promise.all([
    this.projectRepository.createCharacters(projectId, toInsertChars),
    this.projectRepository.createLocations(projectId, toInsertLocs),
    this.projectRepository.createProps(projectId, toInsertProps),
  ]);

  // 5. Asset and Handle Registration
  // Generate handles like @JohnDoe from the newly created entity names
  await this.registerNewEntityHandles([...toInsertChars, ...toInsertLocs, ...toInsertProps], job);

  const [insertedCharacters, insertedLocations, insertedProps] = await Promise.all([
    this.projectRepository.getCharactersByIds(toInsertChars.map(c => c.id)),
    this.projectRepository.getLocationsByIds(toInsertLocs.map(l => l.id)),
    this.projectRepository.getPropsByIds(toInsertProps.map(p => p.id)),
  ]);

  // 6. Visual Context Generation (Parallel Image Creation)
  const [charImages, locImages, propImages] = await Promise.all([
    createGenerateCharacterImagesTool({ context: toolContext }).run({
      characters: insertedCharacters, attempt: job.attempts.currentAttempt
    }),
    createGenerateLocationImagesTool({ context: toolContext }).run({
      locations: insertedLocations, attempt: job.attempts.currentAttempt
    }),
    createGeneratePropImagesTool({ context: toolContext }).run({
      props: insertedProps, attempt: job.attempts.currentAttempt
    }),
  ]);

  // 7. Final Scene Generation with Full KB Context
  const fullContextCharacters = [...resolvedCharacters, ...insertedCharacters];
  const fullContextLocations = [...resolvedLocations, ...insertedLocations];
  const fullContextProps = [...resolvedProps, ...insertedProps];

  const [sceneAttributesResult] = await createGenerateSceneAttributesTool({ context: toolContext }).run([
    {
      partial: {
        ...sceneFields,
        sceneIndex: existingScenes.length,
        characters: fullContextCharacters.map(c => CharacterAttributes.parse(hydrateEntity(c, c.assets))),
        location: fullContextLocations[0] ? LocationAttributes.parse(hydrateEntity(fullContextLocations[0], fullContextLocations[0].assets)) : null,
        props: fullContextProps.map(p => PropAttributes.parse(hydrateEntity(p, p.assets))),
      }
    }
  ]);

  const sceneAttributes = sceneAttributesResult.success ? sceneAttributesResult.output : null;
  if (!sceneAttributes) throw new Error("Failed to generate scene attributes.");

  // 8. Final Persistence and Eventing
  const [finalScene] = await this.projectRepository.createScenes(projectId, [
    mapDomainSceneToInsertScene({
      ...sceneAttributes,
      projectId,
      locationId: fullContextLocations[0]?.id,
      characterReferenceIds: fullContextCharacters.map(c => c.referenceId),
      propReferenceIds: fullContextProps.map(p => p.referenceId),
    })
  ]);

  await this.publishPipelineEvent({
    type: "ENTITY_CREATED",
    projectId,
    payload: [
      ...insertedCharacters.map(c => ({ entityId: c.id, entityType: "character", entity: c })),
      ...insertedLocations.map(l => ({ entityId: l.id, entityType: "location", entity: l })),
      ...insertedProps.map(p => ({ entityId: p.id, entityType: "prop", entity: p })),
      { entityId: finalScene.id, entityType: "scene", entity: finalScene }
    ],
    timestamp: new Date().toISOString(),
  });

  return finalScene;
}
